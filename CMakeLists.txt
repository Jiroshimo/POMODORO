cmake_minimum_required(VERSION 3.16)
project(QuietPomodoro VERSION 1.3.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0)
find_package(Threads REQUIRED)
find_program(GLIB_COMPILE_RESOURCES glib-compile-resources)
if(NOT GLIB_COMPILE_RESOURCES)
  message(FATAL_ERROR "glib-compile-resources is required (included with GLib development tools)")
endif()
set(APP_RESOURCES "${CMAKE_CURRENT_BINARY_DIR}/pomodoro-resources.c")
add_custom_command(OUTPUT "${APP_RESOURCES}"
  COMMAND "${GLIB_COMPILE_RESOURCES}" "${CMAKE_CURRENT_SOURCE_DIR}/assets/resources.xml"
    --sourcedir=${CMAKE_CURRENT_SOURCE_DIR}/assets --generate-source --target=${APP_RESOURCES}
  DEPENDS assets/resources.xml assets/style.css assets/Silkscreen-Regular.ttf assets/Silkscreen-OFL.txt
  VERBATIM)

add_library(pomodoro_audio STATIC src/audio.cpp)
target_include_directories(pomodoro_audio PUBLIC src)
target_link_libraries(pomodoro_audio PUBLIC PkgConfig::GIO Threads::Threads ${CMAKE_DL_LIBS})
add_executable(pomodoro src/main.cpp src/storage.cpp src/font.cpp "${APP_RESOURCES}")
target_include_directories(pomodoro PRIVATE src)
target_link_libraries(pomodoro PRIVATE PkgConfig::GTK3 pomodoro_audio)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(AUDIO_RUNTIME_LIB "audio-runtime/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}")
  target_compile_definitions(pomodoro_audio PRIVATE QUIET_AUDIO_LIBRARY_ARCH="${CMAKE_LIBRARY_ARCHITECTURE}")
  set_target_properties(pomodoro PROPERTIES
    BUILD_RPATH "$ORIGIN/${AUDIO_RUNTIME_LIB};$ORIGIN/${AUDIO_RUNTIME_LIB}/pulseaudio"
    INSTALL_RPATH "$ORIGIN/../lib/quiet-pomodoro/${AUDIO_RUNTIME_LIB};$ORIGIN/../lib/quiet-pomodoro/${AUDIO_RUNTIME_LIB}/pulseaudio")
  install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/audio-runtime/"
    DESTINATION lib/quiet-pomodoro/audio-runtime OPTIONAL
    PATTERN packages EXCLUDE)
endif()
if(WIN32)
  target_link_libraries(pomodoro_audio PUBLIC winmm)
  target_link_libraries(pomodoro PRIVATE gdi32)
  set_target_properties(pomodoro PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
if(UNIX)
  pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
  target_link_libraries(pomodoro PRIVATE PkgConfig::FONTCONFIG)
endif()
if(MSVC)
  target_compile_options(pomodoro PRIVATE /W4 /utf-8)
  target_compile_options(pomodoro_audio PRIVATE /W4 /utf-8)
else()
  target_compile_options(pomodoro PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>")
  target_compile_options(pomodoro_audio PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(timer_tests tests/timer_tests.cpp)
  target_include_directories(timer_tests PRIVATE src)
  add_test(NAME timer COMMAND timer_tests)
  add_executable(storage_tests tests/storage_tests.cpp src/storage.cpp)
  target_include_directories(storage_tests PRIVATE src)
  target_link_libraries(storage_tests PRIVATE PkgConfig::GTK3)
  add_test(NAME storage COMMAND storage_tests)
  add_executable(plan_tests tests/plan_tests.cpp)
  target_include_directories(plan_tests PRIVATE src)
  add_test(NAME plan COMMAND plan_tests)
  add_executable(audio_tests tests/audio_tests.cpp)
  target_link_libraries(audio_tests PRIVATE pomodoro_audio)
  add_test(NAME audio COMMAND audio_tests)
  set_tests_properties(audio PROPERTIES TIMEOUT 10)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    if(EXISTS "${CMAKE_CURRENT_BINARY_DIR}/${AUDIO_RUNTIME_LIB}/libpulse-simple.so.0")
      find_package(Python3 QUIET COMPONENTS Interpreter)
      if(Python3_Interpreter_FOUND)
        add_test(NAME audio_bundled_runtime COMMAND "${Python3_EXECUTABLE}"
          "${CMAKE_CURRENT_SOURCE_DIR}/tests/bundled_audio_test.py" "$<TARGET_FILE:pomodoro>")
        set_tests_properties(audio_bundled_runtime PROPERTIES TIMEOUT 10)
      endif()
    endif()
    add_library(fake_pulse SHARED tests/fake_pulse.cpp)
    set_target_properties(fake_pulse PROPERTIES OUTPUT_NAME pulse-simple SOVERSION 0
      LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/audio-test-libs")
    add_dependencies(audio_tests fake_pulse)
    add_executable(fake_windows_audio tests/fake_windows_audio.cpp)
    target_include_directories(fake_windows_audio PRIVATE src)
    target_link_libraries(fake_windows_audio PRIVATE PkgConfig::GIO)
    set_target_properties(fake_windows_audio PROPERTIES OUTPUT_NAME powershell.exe
      RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/audio-test-bin")
    add_dependencies(audio_tests fake_windows_audio)
    add_test(NAME audio_output COMMAND audio_tests --native-route)
    set_tests_properties(audio_output PROPERTIES TIMEOUT 10
      ENVIRONMENT "LD_LIBRARY_PATH=$<TARGET_FILE_DIR:fake_pulse>;PULSE_SERVER=unix:/tmp/pomodoro-pulse-test")
    foreach(mode route warning failure early-failure no-ack fallback no-ack-fallback both-fail preferred-windows preferred-linux)
      add_test(NAME audio_wsl_${mode} COMMAND audio_tests --wsl-${mode})
      set_tests_properties(audio_wsl_${mode} PROPERTIES TIMEOUT 10 ENVIRONMENT
        "PATH=$<TARGET_FILE_DIR:fake_windows_audio>:$ENV{PATH};WSL_DISTRO_NAME=Pomodoro-Audio-Test;LD_LIBRARY_PATH=$<TARGET_FILE_DIR:fake_pulse>;PULSE_SERVER=unix:/tmp/pomodoro-pulse-test;FAKE_WINDOWS_AUDIO_RESULT=${mode}")
    endforeach()
    set(AUDIO_PATH_TEST_DIR "${CMAKE_CURRENT_BINARY_DIR}/audio-test-tmp/space's-✓")
    file(MAKE_DIRECTORY "${AUDIO_PATH_TEST_DIR}")
    add_test(NAME audio_wsl_path COMMAND audio_tests --wsl-route)
    set_tests_properties(audio_wsl_path PROPERTIES TIMEOUT 10 ENVIRONMENT
      "PATH=$<TARGET_FILE_DIR:fake_windows_audio>:$ENV{PATH};WSL_DISTRO_NAME=Pomodoro-Audio-Test;LD_LIBRARY_PATH=$<TARGET_FILE_DIR:fake_pulse>;PULSE_SERVER=unix:/tmp/pomodoro-pulse-test;TMPDIR=${AUDIO_PATH_TEST_DIR}")
  endif()
  if(UNIX)
    add_executable(assets_tests tests/assets_tests.cpp src/font.cpp "${APP_RESOURCES}")
    target_include_directories(assets_tests PRIVATE src)
    target_link_libraries(assets_tests PRIVATE PkgConfig::GTK3 PkgConfig::FONTCONFIG)
    add_test(NAME assets COMMAND assets_tests)
  endif()
endif()

install(TARGETS pomodoro RUNTIME DESTINATION bin)
if(UNIX AND NOT APPLE)
  install(FILES assets/quiet-pomodoro.desktop DESTINATION share/applications)
  install(FILES assets/quiet-pomodoro.svg DESTINATION share/icons/hicolor/scalable/apps)
endif()
